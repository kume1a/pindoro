# R4a Tab shell and Me Implementation Plan

> **For agentic workers:** Implement this plan one task at a time. Steps use checkbox (`- [ ]`)
> syntax for tracking. The user approved building without stopping between tasks and one commit
> per task (R4a git flow).

**Goal:** Bottom navigation with Free play, Pit stop (due badge) and Me tabs, plus the Me screen.

**Architecture:** A go_router `StatefulShellRoute` with three single-page branches inside a
cross-fading indexed stack; every other route stays on the root navigator and covers the bar. The
bar and stat tile are `pindoro_design` widgets with goldens. Me is a new screen slice
(`features/me`) fed by `meProvider`; the level card moves down to `features/progress/ui` so Home
and Me share it.

**Tech Stack:** Flutter, go_router 17.5, Riverpod 3 (no codegen), Drift, intl, gen-l10n.

**Spec:** `docs/specs/2026-11-07-r4a-tabs-me-design.md`

## Global Constraints

- Design values only from tokens (`PindoroColors`, `PindoroSpace`, `PindoroType`, …); small bundle
  sizes that have no token (56 × 34 pill, 3 / 6 / 10 dp gaps, 18 dp badge) are literals named in a
  doc comment, as `ReviewGauge` does.
- Never uppercase UI strings; every string in `app_en.arb` and `app_ka.arb`; numbers through
  `formatNumber`.
- `apps/mobile` layers: a file imports its own slice or a lower layer (`test/architecture_test.dart`).
- Arrow functions where possible; explicit types; no ticket ids in comments.
- Run only the touched tests locally; goldens only on Linux (`just goldens`).

```dart
import 'tappable.dart';

/// A bottom-nav destination. [badge] counts what waits there (hidden at 0 or
/// null); [semanticLabel] replaces [label] for screen readers.
typedef PindoroNavItem = ({
  PindoroGlyph glyph,
  String label,
  int? badge,
  String? semanticLabel,
});

/// The bottom navigation (`.pd-nav`): `color.surface.raised` under a 2 dp
/// `color.line.base` rule; each item a 56 × 34 icon pill 10 dp from the top
/// (`color.tab.active.bg` on [current]) over its `label.md` label, 3 dp
/// apart. The labels stay above the bottom inset, at least 8 dp under them.
class PindoroNavBar extends StatelessWidget {
  const PindoroNavBar({
    super.key,
    required this.items,
    required this.current,
    required this.onSelect,
  });

  final List<PindoroNavItem> items;
  final int current;
  final ValueChanged<int> onSelect;

  @override
  Widget build(BuildContext context) {
    final c = PindoroTheme.of(context).colors;
    final inset = MediaQuery.paddingOf(context).bottom;
    return DecoratedBox(
      decoration: BoxDecoration(
        color: c.surfaceRaised,
        border: Border(
          top: BorderSide(color: c.lineBase, width: PindoroStroke.base),
        ),
      ),
      child: Padding(
        padding: EdgeInsets.only(
          top: 10,
          bottom: math.max(inset, PindoroSpace.s2),
        ),
        child: Row(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            for (final (i, item) in items.indexed)
              Expanded(
                child: _NavItem(
                  item,
                  selected: i == current,
                  onTap: () => onSelect(i),
                ),
              ),
          ],
        ),
      ),
    );
  }
}

class _NavItem extends StatelessWidget {
  const _NavItem(this.item, {required this.selected, required this.onTap});

  final PindoroNavItem item;
  final bool selected;
  final VoidCallback onTap;

  @override
  Widget build(BuildContext context) {
    final c = PindoroTheme.of(context).colors;
    final ink = selected ? c.tabActiveIcon : c.tabInactiveIcon;
    final pill = selected ? c.tabActiveBg : c.surfaceRaised;
    final badge = item.badge ?? 0;
    return Tappable(
      onTap: onTap,
      selected: selected,
      label: item.semanticLabel ?? item.label,
      child: Column(
        spacing: 3,
        children: [
          Stack(
            clipBehavior: Clip.none,
            children: [
              Container(
                width: 56,
                height: 34,
                alignment: Alignment.center,
                decoration: BoxDecoration(
                  color: pill,
                  borderRadius: BorderRadius.circular(PindoroRadius.md),
                ),
                child: PindoroIcon(item.glyph, color: ink, cut: pill),
              ),
              if (badge > 0)
                PositionedDirectional(top: -6, end: -6, child: _Badge(badge)),
            ],
          ),
          Text(
            item.label,
            textAlign: TextAlign.center,
            style: PindoroType.labelMd.copyWith(color: ink),
          ),
        ],
      ),
    );
  }
}

/// The due count: an 18 dp cobalt pill with a 2 dp `color.surface.raised`
/// ring that lifts it off the icon pill.
```

```dart
/// One number and its label (`.pd-stattile`), for lesson complete and Me:
/// a [LipBox] with `radius.lg`, an optional 20 dp [glyph] (default colour
/// `color.primary.text`) 6 dp before the `label.md` label, the value in
/// `numeric.md`. Reads as one label, "[label], [value]" unless
/// [semanticLabel] says otherwise.
class StatTile extends StatelessWidget {
  const StatTile({
    super.key,
    required this.label,
    required this.value,
    this.glyph,
    this.glyphColor,
    this.semanticLabel,
    this.onTap,
  });

  final String label;
  final String value;
  final PindoroGlyph? glyph;
  final Color? glyphColor;
  final String? semanticLabel;
  final VoidCallback? onTap;

  @override
  Widget build(BuildContext context) {
    final c = PindoroTheme.of(context).colors;
    final glyph = this.glyph;
    final tile = LipBox(
      border: c.lineBase,
      radius: PindoroRadius.lg,
      padding: const EdgeInsets.all(PindoroSpace.s3),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        spacing: 2,
        children: [
          Row(
            spacing: 6,
            children: [
              if (glyph != null)
                PindoroIcon(
                  glyph,
                  size: PindoroSize.iconSm,
                  color: glyphColor ?? c.primaryText,
                ),
              Flexible(
                child: Text(
                  label,
                  style: PindoroType.labelMd.copyWith(color: c.textSecondary),
                ),
              ),
            ],
          ),
          Text(
            value,
            style: PindoroType.numericMd.copyWith(color: c.textPrimary),
          ),
        ],
      ),
    );
    final a11y = semanticLabel ?? '$label, $value';
    return onTap == null
        ? Semantics(
            container: true,
            label: a11y,
            excludeSemantics: true,
            child: tile,
          )
        : Tappable(onTap: onTap, label: a11y, child: tile);
  }
}
```

```dart
/// An [IndexedStack] that cross-fades to [index] (`duration.fast`); every
/// child keeps its state, the hidden ones offstage with tickers off.
/// Instant with reduce motion on.
class FadeIndexedStack extends StatefulWidget {
  const FadeIndexedStack({
    super.key,
    required this.index,
    required this.children,
  });

  final int index;
  final List<Widget> children;

  @override
  State<FadeIndexedStack> createState() => _FadeIndexedStackState();
}

class _FadeIndexedStackState extends State<FadeIndexedStack>
    with SingleTickerProviderStateMixin {
  late final AnimationController _fade = AnimationController(
    vsync: this,
    duration: PindoroMotion.durationFast,
    value: 1,
  )..addStatusListener(_settled);

  /// The child fading out, until the fade ends.
  int? _leaving;

  void _settled(AnimationStatus status) {
    if (status.isCompleted) setState(() => _leaving = null);
  }

  @override
  void didUpdateWidget(FadeIndexedStack old) {
    super.didUpdateWidget(old);
    if (old.index == widget.index) return;
    if (_calm(context)) {
      _leaving = null;
      _fade.value = 1;
      return;
    }
    _leaving = old.index;
    unawaited(_fade.forward(from: 0));
  }

  @override
  void dispose() {
    _fade.dispose();
    super.dispose();
  }

  Animation<double> _opacity(int i) => i == widget.index
      ? _fade
      : i == _leaving
      ? ReverseAnimation(_fade)
      : kAlwaysCompleteAnimation;

  @override
  Widget build(BuildContext context) => Stack(
    fit: StackFit.expand,
    children: [
      for (final (i, child) in widget.children.indexed)
        Offstage(
          offstage: i != widget.index && i != _leaving,
          child: TickerMode(
            enabled: i == widget.index,
            child: IgnorePointer(
              ignoring: i != widget.index,
              child: FadeTransition(opacity: _opacity(i), child: child),
            ),
          ),
        ),
    ],
  );
}
```

```dart
typedef MeData = ({

  DateTime? since,
  int bestStreak,

  /// The chosen city first, then the other played ones.
  List<MeCity> cities,

/// Me's numbers. Each city is counted in its own pack: the open one, or its
/// installed pack opened just for the count; a played city whose pack isn't
/// on the phone is left out.
final meProvider = FutureProvider.autoDispose<MeData>((ref) async {
  ref.watch(progressChangesProvider);
  final dao = ref.watch(progressDaoProvider);
  final store = ref.watch(packStoreProvider);
  final open = await ref.watch(packRepositoryProvider.future);
  final openStreets = await ref.watch(cityStreetsProvider.future);
  final streak = await ref.watch(streakProvider.future);

  Future<MeCity> count(
    PackRepository pack,
    Map<String, NamedStreet> streets,
  ) async {
    final city = pack.city();
    return (
      id: city.id,
      nameKa: city.nameKa,
      nameEn: city.nameEn,
      count: (await dao.knownIn(pack, streets)).city,
    );
  }

  final cities = [await count(open, openStreets)];
  for (final id in await dao.recentCityIds()) {
    final file = id == open.city().id ? null : store.newest(id);
    if (file == null) continue;
    final pack = SqlitePackRepository.open(file.path);
    try {
      cities.add(await count(pack, streetsByPartId(pack.streets())));
    } finally {
      pack.close();
    }
  }
  return (
    totalXp: await dao.totalXp(),
    since: await dao.firstRoundAt(),
    bestStreak: streak.summary.best,
    cities: cities,
  );
});
```

```dart
/// The bottom navigation around the tabs (canvas R4-14): Free play, the Pit
/// stop with its due count, Me. Back on another tab goes to Free play first;
/// tapping the open tab scrolls it to the top.
class TabShell extends ConsumerWidget {
  const TabShell({super.key, required this.shell, required this.scroll});

  static const int tabs = 3;

  final StatefulNavigationShell shell;

  /// Each tab page's primary scroll controller.
  final List<ScrollController> scroll;

  void _select(BuildContext context, WidgetRef ref, int index) {
    ref.read(feedbackProvider).tap();
    if (index != shell.currentIndex) {
      shell.goBranch(index);
    } else if (scroll[index].hasClients) {
      _toTop(context, scroll[index]);
    }
  }

  void _toTop(BuildContext context, ScrollController controller) {
    if (MediaQuery.disableAnimationsOf(context)) return controller.jumpTo(0);
    unawaited(
      controller.animateTo(
        0,
        duration: PindoroMotion.durationBase,
        curve: PindoroMotion.easeDecelerate,
      ),
    );
  }

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final l10n = context.l10n;
    final due = ref.watch(pitStopProvider).value?.due.length ?? 0;
    return PopScope(
      canPop: shell.currentIndex == 0,
      onPopInvokedWithResult: (didPop, _) {
        if (!didPop) shell.goBranch(0);
      },
      child: Scaffold(
        body: shell,
        bottomNavigationBar: PindoroNavBar(
          current: shell.currentIndex,
          onSelect: (index) => _select(context, ref, index),
          items: [
            (
              glyph: PindoroGlyph.map,
              label: l10n.tabFreePlay,
              badge: null,
              semanticLabel: null,

              glyph: PindoroGlyph.user,
              label: l10n.meTitle,
              badge: null,
              semanticLabel: null,
            ),
          ],
        ),

    );
```
