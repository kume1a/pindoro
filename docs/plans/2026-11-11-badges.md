# Badges Implementation Plan

> **For agentic workers:** Implement this plan one task at a time, pausing for review between tasks. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The design system's eight starter badges: rules in pindoro_core, the shield and tile in
pindoro_design, saved badges and the stats behind them in the app, a shelf on Me and a reward
sheet on the end screens.

**Architecture:** pindoro_core `badgeProgress(id, stats)` is the only rule. The app reads
`BadgeStats` fresh from the database and the installed packs of played cities (never from cached
providers, never by loading a pack), saves each newly earned badge in `badge` (insert or ignore),
and returns the new ids to the end screen that awarded them. Me shows every badge with its
progress; the New pill comes from `seen_at`.

**Tech Stack:** Dart 3 / Flutter, Riverpod 3 (no codegen), Drift 2.35 (dates stored as text),
go_router `StatefulShellRoute`, gen-l10n.

## Global Constraints

- Spec: `docs/specs/2026-11-11-badges-design.md`.
- Never uppercase UI strings; FiraGO and tokens only; no literal colours in widgets.
- Generated code (tokens, l10n, Drift) is never committed; `just gen` after token, ARB or table
  changes. Commit the new drift dump `drift_schema_v11.json`.
- Goldens are Linux-only: regenerate with `just goldens`, commit the PNGs.
- Run only the touched tests locally (the user's rule); CI runs the rest.
- Commits only after the user approves; no co-author trailer; branch `badges`.

```dart
import 'dart:math';

/// The starter badges (design system component Badge), in shelf order.
enum BadgeId {
  firstPin,
  bullseye,
  onFire,
  navigator,
  roadTrip,
  nightOwl,
  explorer,
  pitCrew,
}

/// What badges are measured on, gathered from the player's progress.
typedef BadgeStats = ({
  /// Pins dropped in rounds and lesson pin steps.
  int pins,

  /// Of those, Perfect.
  int perfects,
  int bestStreak,

  /// Raions with every street known, in every city.
  int masteredDistricts,

  /// The most raions with a known street in one city.
  int knownDistricts,

  /// Road trips answered: lesson trips and Final drive trips.
  int roadTrips,

  /// Any answer or lesson finished before [BadgeRules.lateBefore] local time.
  bool lateAnswer,

  /// Streets cleared from the Pit stop.
  int pitStopCleared,
});

/// Where a badge stands; [value] never shows past [target].
typedef BadgeProgress = ({int value, int target});

extension BadgeProgressRules on BadgeProgress {
  bool get earned => value >= target;
}

abstract final class BadgeRules {
  /// Night owl: an answer from midnight until this local hour.
  static const int lateBefore = 4;
}

BadgeProgress badgeProgress(BadgeId id, BadgeStats s) {
  final (value, target) = switch (id) {
    BadgeId.firstPin => (s.pins, 1),
    BadgeId.bullseye => (s.perfects, 10),
    BadgeId.onFire => (s.bestStreak, 7),
    BadgeId.navigator => (s.masteredDistricts, 1),
    BadgeId.roadTrip => (s.roadTrips, 5),
    BadgeId.nightOwl => (s.lateAnswer ? 1 : 0, 1),
    BadgeId.explorer => (s.knownDistricts, 5),
    BadgeId.pitCrew => (s.pitStopCleared, 50),
  };
  return (value: min(value, target), target: target);
}
```

```dart
import 'package:flutter/material.dart';

import '../generated/pindoro_tokens.dart';
import '../theme.dart';
import 'district_map.dart';
import 'graphics.dart';
import 'surfaces.dart';

/// A badge's shield (component Badge): a `color.badge.rim` rim, the coloured
/// [field] inset in it and one [glyph] in white. [locked] draws the locked
/// stop's colours and a lock instead. [size] is its width; it is 100 / 88 as
/// tall.
class BadgeShield extends StatelessWidget {
  const BadgeShield({
    super.key,
    required this.glyph,
    required this.field,
    this.locked = false,
    this.size = 76,
  });

  final PindoroGlyph glyph;
  final Color field;
  final bool locked;
  final double size;

  @override
  Widget build(BuildContext context) {
    final c = PindoroTheme.of(context).colors;
    final face = locked ? c.pathNodeLockedFill : field;
    // The design draws the shield in an 88 × 100 box.
    final k = size / 88;
    return SizedBox(
      width: size,
      height: 100 * k,
      child: CustomPaint(
        painter: _ShieldPainter(
          rim: locked ? c.pathNodeLockedLip : c.badgeRim,
          field: face,
        ),
        child: Stack(
          children: [
            // The design's glyph: 24 dp at 1.5 ×, from (26, 28).
            Positioned(
              left: 26 * k,
              top: 28 * k,
              child: PindoroIcon(
                locked ? PindoroGlyph.lock : glyph,
                size: 36 * k,
                color: locked ? c.pathNodeLockedIcon : c.textOnColor,
                cut: face,
              ),
            ),
          ],
        ),
      ),
    );
  }
}

class _ShieldPainter extends CustomPainter {
  _ShieldPainter({required this.rim, required this.field});

  final Color rim;
  final Color field;

  // The design's two shield paths (viewBox 88 × 100), scaled by k.
  static Path _outer(double k) => Path()
    ..moveTo(44 * k, 2 * k)
    ..lineTo(84 * k, 18 * k)
    ..lineTo(84 * k, 48 * k)
    ..cubicTo(84 * k, 74 * k, 66 * k, 90 * k, 44 * k, 98 * k)
    ..cubicTo(22 * k, 90 * k, 4 * k, 74 * k, 4 * k, 48 * k)
    ..lineTo(4 * k, 18 * k)
    ..close();

  static Path _inner(double k) => Path()
    ..moveTo(44 * k, 10 * k)
    ..lineTo(76 * k, 23 * k)
    ..lineTo(76 * k, 48 * k)
    ..cubicTo(76 * k, 69 * k, 62 * k, 82 * k, 44 * k, 89 * k)
    ..cubicTo(26 * k, 82 * k, 12 * k, 69 * k, 12 * k, 48 * k)
    ..lineTo(12 * k, 23 * k)
    ..close();

  @override
  void paint(Canvas canvas, Size size) {
    final k = size.width / 88;
    canvas
      ..drawPath(_outer(k), Paint()..color = rim)
      ..drawPath(_inner(k), Paint()..color = field);
  }

  @override
  bool shouldRepaint(_ShieldPainter old) =>
      old.rim != rim || old.field != field;
}

/// A badge on the shelf (component Badge): its 56 dp shield, name and
/// condition on a raised card. A locked badge counting toward its target
/// shows [progress] (0–1) in a mini bar; [newLabel] shows the New pill.
/// Reads as one label, [semanticLabel].
class BadgeTile extends StatelessWidget {
  const BadgeTile({
    super.key,
    required this.glyph,
    required this.field,
    required this.name,
    required this.condition,
    required this.semanticLabel,
    this.locked = false,
    this.progress,
    this.newLabel,
  });

  final PindoroGlyph glyph;
  final Color field;
  final String name;
  final String condition;
  final String semanticLabel;
  final bool locked;
  final double? progress;
  final String? newLabel;

  @override
  Widget build(BuildContext context) {
    final c = PindoroTheme.of(context).colors;
    final progress = this.progress;
    final newLabel = this.newLabel;
    return Semantics(
      container: true,
      label: semanticLabel,
      excludeSemantics: true,
      child: Stack(
        children: [
          LipBox(
            border: c.lineBase,
            radius: PindoroRadius.lg,
            padding: const EdgeInsets.symmetric(
              horizontal: PindoroSpace.s2,
              vertical: PindoroSpace.s3,
            ),
            child: Column(
              spacing: 2,
              children: [
                BadgeShield(
                  glyph: glyph,
                  field: field,
                  locked: locked,
                  size: 56,
                ),
                const SizedBox(height: PindoroSpace.s1),
                Text(
                  name,
                  textAlign: TextAlign.center,
                  style: PindoroType.titleSm.copyWith(
                    color: locked ? c.textSecondary : c.textPrimary,
                  ),
                ),
                Text(
                  condition,
                  textAlign: TextAlign.center,
                  style: PindoroType.bodySm.copyWith(color: c.textSecondary),
                ),
                if (progress != null)
                  Padding(
                    padding: const EdgeInsets.only(top: PindoroSpace.s1),
                    child: Minibar(value: progress),
                  ),
              ],
            ),
          ),
          if (newLabel != null)
            PositionedDirectional(
              top: PindoroSpace.s2,
              end: PindoroSpace.s2,
              child: Container(
                // The design's pill: 1 × 7 dp around label.sm.
                padding: const EdgeInsets.symmetric(horizontal: 7, vertical: 1),
                decoration: BoxDecoration(
                  color: c.accentBase,
                  borderRadius: BorderRadius.circular(PindoroRadius.full),
                ),
                child: Text(
                  newLabel,
                  style: PindoroType.labelSm.copyWith(color: c.accentOn),
                ),
              ),
            ),
        ],
      ),
    );
  }
}
```

```dart
      'SELECT '

      "(SELECT COUNT(*) FROM answer WHERE tier = 'perfect') + "
      '(SELECT COUNT(*) FROM lesson_answer '

      "WHERE kind != 'placed' AND ${hour('finished_at')} < ?1) AS late, "
```

```dart
/// The player's [BadgeStats], read fresh from the database and the installed
/// packs of played cities: an end screen awards right after its play was
/// saved, before any provider reloads. Never loads or downloads a pack.
Future<BadgeStats> readBadgeStats(ProgressDao dao, PackStore store) async {
  final counts = await dao.badgeCounts();
  final raions = [
    for (final city in await dao.recentCityIds())
      ?await readInstalledPack(store, city, (pack, streets) async {
        final ids = {
          for (final d in pack.districts())
            if (d.kind == DistrictKind.district) d.id,
        };
        final known = (await dao.knownIn(pack, streets)).byDistrict;
        final inRaions = [
          for (final MapEntry(:key, :value) in known.entries)
            if (ids.contains(key)) value,
        ];
        return (
          mastered: inRaions.where((c) => c.total > 0 && c.known >= c.total).length,
          known: inRaions.where((c) => c.known > 0).length,
        );
      }),
  ];
  final days = (await dao.goalDayRounds()).keys.toSet();
  return (
    pins: counts.pins,
    perfects: counts.perfects,
    bestStreak: streakOf(days, calendarDay(DateTime.now())).best,
    masteredDistricts: raions.fold(0, (sum, r) => sum + r.mastered),
    knownDistricts: raions.fold(0, (most, r) => max(most, r.known)),
    roadTrips: await dao.roadTrips(),
    lateAnswer: counts.lateAnswer,
    pitStopCleared: counts.pitStopCleared,
  );
}
```

```dart
/// The reward moment's shield (component Badge).
const double _rewardShield = 110;

/// One reward sheet per badge in [ids], one after another; closing one any
/// way marks it seen.
Future<void> showBadgeRewards(
  BuildContext context,
  WidgetRef ref,
  List<BadgeId> ids,
) async {
  for (final id in ids) {
    if (!context.mounted) return;
    ref.read(feedbackProvider).badge();
    await showModalBottomSheet<void>(
      context: context,
      useRootNavigator: true,
      useSafeArea: true,
      isScrollControlled: true,
      backgroundColor: PindoroTheme.of(context).colors.rewardSoft,
      builder: (_) => _BadgeRewardSheet(id),
    );
    await ref.read(progressDaoProvider).seeBadges([id], DateTime.now());
  }
}

/// For an end screen: awards this play's badges as it opens (its play is
/// saved by then) and shows their reward sheets once [showBadgesAfter] is
/// first called, after the arrival and any level-up moment.
mixin BadgeMoments<T extends ConsumerStatefulWidget> on ConsumerState<T> {
  late final Future<List<BadgeId>> _earned;
  bool _scheduled = false;

  @override
  void initState() {
    super.initState();
    // A failed award (a pack that won't open) tries again at the next end
    // screen; the screen itself never fails for it.
    _earned = ref.read(awardBadgesProvider)().onError((_, _) => const []);
  }

  void showBadgesAfter(Duration delay) {
    if (_scheduled) return;
    _scheduled = true;
    unawaited(() async {
      final ids = await _earned;
      if (ids.isEmpty) return;
      await Future<void>.delayed(delay);
      if (mounted) await showBadgeRewards(context, ref, ids);
    }());
  }
}
```
